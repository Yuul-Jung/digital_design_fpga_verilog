## 학습 안내

4장에서 만든 Tick·BCD 카운터·FND 스캔을 **기능별 모듈로 나누고 최상위(Top) 모듈에서 연결**해, 실제로 쓸 수 있는 두 가지 시스템을 완성합니다.

| 과제 | 표시 형식 | 기준 Tick | 표시 범위 |
|---|---|---|---|
| 스톱워치 | `SS.HH` (초.백분의 1초) | 0.01초 | 00.00 ~ 99.99 |
| 24시간 디지털 시계 | `HH.MM` (시.분) | 1초 → 1분 | 00.00 ~ 23.59 |

**구조화 설계 원칙**

| 원칙 | 내용 |
|---|---|
| 하나의 모듈 = 하나의 기능 | Tick 생성, 카운터, 제어, 스캔, MUX, 디코더를 분리 |
| Top은 연결 중심 | 기능 로직은 하위 모듈에, Top에는 인스턴스와 간단한 결합식만 |
| 단일 클록 | 모든 레지스터는 `always @(posedge clk)` |
| Tick은 Enable | 0.01초·1초·1분 Tick 모두 1클록 폭 Enable 신호 |
| 모듈 단위 검증 | 카운터·제어 모듈마다 테스트벤치로 먼저 확인 |

## 스톱워치 Tick과 카운터

**RTL 1 · 0.01초 Tick** — 100MHz에서 0.01초는 1,000,000클록(비교값 **999,999**)입니다. `enable`이 0이면 기준 카운트를 0으로 되돌려, 다시 시작할 때 0.01초 단위가 깨끗하게 맞춰지도록 합니다.

파일명: `clock_divider_10ms.v`

```verilog
module clock_divider_10ms(
    input      clk,     // 100MHz 기준 Clock
    input      reset,   // 동기 Reset
    input      enable,  // Tick 생성 허용
    output reg tick     // 0.01초마다 1 Clock 동안 1
);

    reg [19:0] count_div;   // 1,000,000 Clock Counter

    always @(posedge clk) begin
        if (reset) begin
            count_div <= 20'd0;
            tick      <= 1'b0;
        end
        else begin
            tick <= 1'b0;                       // Tick 기본값은 0
            if (enable) begin                   // Enable 상태에서만 Count
                if (count_div == 20'd999_999) begin
                    count_div <= 20'd0;         // 1,000,000 Clock 도달
                    tick      <= 1'b1;          // 1 Clock 폭 Tick
                end
                else begin
                    count_div <= count_div + 1'b1;
                end
            end
            else begin
                count_div <= 20'd0;             // 정지 시 기준 Count 초기화
            end
        end
    end

endmodule
```

**RTL 2 · 스톱워치 카운터** — 4장의 중첩 `if` 대신 **현재 값으로 Carry 조건을 미리 계산**해 각 자리를 독립적으로 갱신합니다. 자리 수가 늘어도 구조가 단순하게 유지됩니다.

파일명: `stopwatch_counter_4digit.v`

```verilog
module stopwatch_counter_4digit(
    input            clk,     // 기준 Clock
    input            reset,   // 동기 Reset
    input            enable,  // Count 증가 Enable
    output reg [3:0] digit0,  // 0.01초 자리
    output reg [3:0] digit1,  // 0.1초 자리
    output reg [3:0] digit2,  // 1초 자리
    output reg [3:0] digit3   // 10초 자리
);

    wire carry0;   // digit0 자리 올림 조건
    wire carry1;   // digit1 자리 올림 조건
    wire carry2;   // digit2 자리 올림 조건

    // 현재 Digit 값 기준 Carry 조건 생성
    assign carry0 = (digit0 == 4'd9);
    assign carry1 = (digit0 == 4'd9) && (digit1 == 4'd9);
    assign carry2 = (digit0 == 4'd9) && (digit1 == 4'd9) && (digit2 == 4'd9);

    always @(posedge clk) begin
        if (reset) begin
            digit0 <= 4'd0;
            digit1 <= 4'd0;
            digit2 <= 4'd0;
            digit3 <= 4'd0;
        end
        else begin
            if (enable) begin   // 0.01초 Tick에서만 증가
                digit0 <= carry0 ? 4'd0 : digit0 + 1'b1;
                if (carry0)
                    digit1 <= (digit1 == 4'd9) ? 4'd0 : digit1 + 1'b1;
                if (carry1)
                    digit2 <= (digit2 == 4'd9) ? 4'd0 : digit2 + 1'b1;
                if (carry2)
                    digit3 <= (digit3 == 4'd9) ? 4'd0 : digit3 + 1'b1;
            end
        end
    end

endmodule
```

파일명: `tb_stopwatch_counter_4digit.v`

```verilog
`timescale 1ns / 1ps
module tb_stopwatch_counter_4digit;

    reg        clk;      // Testbench Clock
    reg        reset;    // Reset 입력
    reg        tick;     // Counter Enable 입력
    wire [3:0] digit0;   // 0.01초 자리
    wire [3:0] digit1;   // 0.1초 자리
    wire [3:0] digit2;   // 1초 자리
    wire [3:0] digit3;   // 10초 자리

    stopwatch_counter_4digit dut(
        .clk(clk),
        .reset(reset),
        .enable(tick),
        .digit0(digit0),
        .digit1(digit1),
        .digit2(digit2),
        .digit3(digit3)
    );

    // 100MHz Clock 생성
    initial begin
        clk = 1'b0;
        forever #5 clk = ~clk;
    end

    // 1 Clock 폭 Tick 생성 Task
    task make_tick;
        begin
            tick = 1'b1; #10;
            tick = 1'b0; #20;
        end
    endtask

    initial begin
        reset = 1'b1; tick = 1'b0; #20;
        reset = 1'b0;
        repeat (125) begin      // 00.00 → 01.25
            make_tick;
        end
        $finish;
    end

endmodule
```

125번의 Tick 후 표시값은 `01.25`(digit3=0, digit2=1, digit1=2, digit0=5)가 되어야 합니다.

## 스톱워치 제어와 FND 모듈

**RTL 3 · Start/Stop 제어** — 버튼을 누른 순간의 명령을 `running` 상태로 기억합니다. Start가 Stop보다 우선합니다.

파일명: `start_stop_control.v`

```verilog
module start_stop_control(
    input      clk,      // 기준 Clock
    input      reset,    // 동기 Reset
    input      start,    // Start 입력
    input      stop,     // Stop 입력
    output reg running   // 실행 상태
);

    always @(posedge clk) begin
        if (reset) begin
            running <= 1'b0;        // Reset 시 정지
        end
        else begin
            if (start)
                running <= 1'b1;    // Start 입력 시 실행 상태 저장
            else if (stop)
                running <= 1'b0;    // Stop 입력 시 정지 상태 저장
        end
    end

endmodule
```

파일명: `tb_start_stop_control.v`

```verilog
`timescale 1ns / 1ps
module tb_start_stop_control;

    reg  clk, reset, start, stop;
    wire running;

    start_stop_control dut(
        .clk(clk), .reset(reset), .start(start), .stop(stop), .running(running)
    );

    initial begin
        clk = 1'b0;
        forever #5 clk = ~clk;
    end

    initial begin
        reset = 1'b1; start = 1'b0; stop = 1'b0; #20;
        reset = 1'b0;
        start = 1'b1; #10; start = 1'b0; #100;   // running = 1
        stop  = 1'b1; #10; stop  = 1'b0; #100;   // running = 0
        start = 1'b1; #10; start = 1'b0; #50;    // running = 1
        reset = 1'b1; #20;                       // running = 0
        $finish;
    end

endmodule
```

**RTL 4 · FND 스캔 카운터**

파일명: `fnd_scan_counter.v`

```verilog
module fnd_scan_counter(
    input            clk,        // 기준 Clock
    input            reset,      // 동기 Reset
    output reg [1:0] digit_sel   // 현재 선택 Digit
);

    reg [15:0] scan_div;   // FND Scan 속도 생성 Counter

    always @(posedge clk) begin
        if (reset) begin
            scan_div  <= 16'd0;
            digit_sel <= 2'd0;
        end
        else begin
            scan_div  <= scan_div + 1'b1;     // Scan Counter 증가
            digit_sel <= scan_div[15:14];     // 4개 Digit 순환 선택
        end
    end

endmodule
```

**RTL 5 · FND MUX + DP** — 1초 자리(digit2)를 표시하는 순간에만 DP를 켜서 `SS.HH`의 구분점을 만듭니다.

파일명: `fnd_mux_4digit_dp.v`

```verilog
module fnd_mux_4digit_dp(
    input      [3:0] digit0,         // 0.01초 자리
    input      [3:0] digit1,         // 0.1초 자리
    input      [3:0] digit2,         // 1초 자리
    input      [3:0] digit3,         // 10초 자리
    input      [1:0] digit_sel,      // 현재 선택 Digit
    output reg [3:0] current_digit,  // Decoder로 보낼 현재 Digit
    output reg [3:0] an,             // FND Digit 선택, Active-Low
    output reg       dp              // Decimal Point, Active-Low
);

    always @(*) begin
        current_digit = 4'd0;   // 기본 표시값
        an            = 4'b1111;// 기본은 모든 Digit OFF
        dp            = 1'b1;   // 기본은 DP OFF

        case (digit_sel)
            2'd0: begin current_digit = digit0; an = 4'b1110; dp = 1'b1; end
            2'd1: begin current_digit = digit1; an = 4'b1101; dp = 1'b1; end
            2'd2: begin current_digit = digit2; an = 4'b1011; dp = 1'b0; end  // DP ON
            2'd3: begin current_digit = digit3; an = 4'b0111; dp = 1'b1; end
        endcase
    end

endmodule
```

**RTL 6 · FND 디코더**

파일명: `fnd_decoder.v`

```verilog
module fnd_decoder(
    input      [3:0] bcd,   // 표시할 BCD 값
    output reg [6:0] seg    // Segment 출력, Active-Low
);

    always @(*) begin
        case (bcd)
            4'd0:    seg = 7'b1000000;
            4'd1:    seg = 7'b1111001;
            4'd2:    seg = 7'b0100100;
            4'd3:    seg = 7'b0110000;
            4'd4:    seg = 7'b0011001;
            4'd5:    seg = 7'b0010010;
            4'd6:    seg = 7'b0000010;
            4'd7:    seg = 7'b1111000;
            4'd8:    seg = 7'b0000000;
            4'd9:    seg = 7'b0010000;
            default: seg = 7'b1111111;   // 예외값은 표시하지 않음
        endcase
    end

endmodule
```

## Basys3 스톱워치 Top

| 포트 | 장치 | 역할 |
|---|---|---|
| `clk` | W5 | 100MHz 클록 |
| `btnC` | 가운데 버튼 | Start |
| `btnU` | 위 버튼 | Stop |
| `btnD` | 아래 버튼 | Reset |
| `sw[0]` | SW0 | 전체 Enable |
| `led[0]` | LED0 | `running` 상태 |
| `led[4:1]` | LED1~4 | 0.01초 자리 |
| `seg`, `an`, `dp` | FND | `SS.HH` 표시 |

파일명: `basys3_step3_struct_stopwatch.v`

```verilog
module basys3_step3_struct_stopwatch(
    input         clk,    // Basys3 100MHz Clock
    input         btnC,   // Start 버튼
    input         btnU,   // Stop 버튼
    input         btnD,   // Reset 버튼
    input  [15:0] sw,     // sw[0]: 전체 Enable
    output [15:0] led,    // 상태 확인 LED
    output [6:0]  seg,    // Segment 출력, Active-Low
    output [3:0]  an,     // Digit 선택 출력, Active-Low
    output        dp      // Decimal Point, Active-Low
);

    wire       tick_0p01s;     // 0.01초 Tick
    wire       running;        // 실행 상태
    wire       count_enable;   // Counter 증가 Enable
    wire [3:0] digit0;         // 0.01초 자리
    wire [3:0] digit1;         // 0.1초 자리
    wire [3:0] digit2;         // 1초 자리
    wire [3:0] digit3;         // 10초 자리
    wire [1:0] digit_sel;      // FND 선택 Digit
    wire [3:0] current_digit;  // 현재 표시 Digit

    // Top 내부 제어 신호
    assign count_enable = sw[0] && running && tick_0p01s;

    // LED 확인 신호
    assign led[0]    = running;
    assign led[4:1]  = digit0;
    assign led[15:5] = 11'd0;

    // 0.01초 Tick 생성
    clock_divider_10ms u_tick(
        .clk(clk),
        .reset(btnD),
        .enable(sw[0] && running),
        .tick(tick_0p01s)
    );

    // Start / Stop 상태 제어
    start_stop_control u_control(
        .clk(clk),
        .reset(btnD),
        .start(btnC),
        .stop(btnU),
        .running(running)
    );

    // Stopwatch Counter
    stopwatch_counter_4digit u_counter(
        .clk(clk),
        .reset(btnD),
        .enable(count_enable),
        .digit0(digit0),
        .digit1(digit1),
        .digit2(digit2),
        .digit3(digit3)
    );

    // FND Scan Counter
    fnd_scan_counter u_scan(
        .clk(clk),
        .reset(btnD),
        .digit_sel(digit_sel)
    );

    // FND Digit 선택 및 DP 제어
    fnd_mux_4digit_dp u_mux(
        .digit0(digit0),
        .digit1(digit1),
        .digit2(digit2),
        .digit3(digit3),
        .digit_sel(digit_sel),
        .current_digit(current_digit),
        .an(an),
        .dp(dp)
    );

    // FND Segment Decoder
    fnd_decoder u_decoder(
        .bcd(current_digit),
        .seg(seg)
    );

endmodule
```

## 시계 Tick과 시간 카운터

**RTL 1 · 1초 Tick** — 100MHz × 1초 = 100,000,000클록(비교값 **99,999,999**, 27비트)

파일명: `clock_divider_1s.v`

```verilog
module clock_divider_1s(
    input      clk,      // 100MHz 기준 Clock
    input      reset,    // 동기 Reset
    input      enable,   // Tick 생성 허용
    output reg tick_1s   // 1초마다 1 Clock 동안 1
);

    reg [26:0] count_div;   // 100,000,000 Clock Count용

    always @(posedge clk) begin
        if (reset) begin
            count_div <= 27'd0;
            tick_1s   <= 1'b0;
        end
        else begin
            tick_1s <= 1'b0;
            if (enable) begin
                if (count_div == 27'd99_999_999) begin
                    count_div <= 27'd0;     // 1초 도달
                    tick_1s   <= 1'b1;      // 1 Clock 폭 Tick
                end
                else begin
                    count_div <= count_div + 1'b1;
                end
            end
            else begin
                count_div <= 27'd0;         // 정지 시 Count 초기화
            end
        end
    end

endmodule
```

**RTL 2 · 60초 카운터** — 1초 Tick을 60번 세어 1분 Tick을 만듭니다.

파일명: `second_counter_60.v`

```verilog
module second_counter_60(
    input            clk,        // 기준 Clock
    input            reset,      // 동기 Reset
    input            tick_1s,    // 1초 Tick Enable
    output reg       tick_1min,  // 60초마다 1 Clock 동안 1
    output reg [5:0] sec_count   // 내부 초 Count, 0~59
);

    always @(posedge clk) begin
        if (reset) begin
            sec_count <= 6'd0;
            tick_1min <= 1'b0;
        end
        else begin
            tick_1min <= 1'b0;
            if (tick_1s) begin
                if (sec_count == 6'd59) begin
                    sec_count <= 6'd0;      // 59초 다음 0초
                    tick_1min <= 1'b1;      // 1분 Tick 발생
                end
                else begin
                    sec_count <= sec_count + 1'b1;
                end
            end
        end
    end

endmodule
```

**RTL 3 · HH:MM 카운터** — 분은 00~59, 시는 00~23입니다. **23:59 다음 00:00**을 별도 조건(`last_time`)으로 처리하는 것이 핵심입니다.

| 조건 | 동작 |
|---|---|
| `min_ones == 9` | 1분 자리 0, 10분 자리 증가 |
| `min_tens == 5 && min_ones == 9` | 분 00, 시 증가 |
| `hour_ones == 9` 이면서 시 증가 | 09:59 → 10:00, 19:59 → 20:00 |
| 23:59 | 00:00 |

파일명: `clock_counter_hhmm.v`

```verilog
module clock_counter_hhmm(
    input            clk,        // 기준 Clock
    input            reset,      // 동기 Reset
    input            tick_1min,  // 1분 Tick Enable
    output reg [3:0] min_ones,   // 1분 자리, 0~9
    output reg [3:0] min_tens,   // 10분 자리, 0~5
    output reg [3:0] hour_ones,  // 1시간 자리
    output reg [3:0] hour_tens   // 10시간 자리, 0~2
);

    wire carry_min_ones;   // 1분 자리 Carry 조건
    wire carry_min_tens;   // 59분 Carry 조건
    wire last_time;        // 23:59 조건

    // 현재 시간값 기준 Carry 조건 생성
    assign carry_min_ones = (min_ones == 4'd9);
    assign carry_min_tens = (min_ones == 4'd9) && (min_tens == 4'd5);
    assign last_time      = (hour_tens == 4'd2) && (hour_ones == 4'd3) &&
                            (min_tens  == 4'd5) && (min_ones  == 4'd9);

    always @(posedge clk) begin
        if (reset) begin
            min_ones  <= 4'd0;
            min_tens  <= 4'd0;
            hour_ones <= 4'd0;
            hour_tens <= 4'd0;
        end
        else begin
            if (tick_1min) begin
                if (last_time) begin            // 23:59 다음 00:00
                    min_ones  <= 4'd0;
                    min_tens  <= 4'd0;
                    hour_ones <= 4'd0;
                    hour_tens <= 4'd0;
                end
                else begin
                    min_ones <= carry_min_ones ? 4'd0 : min_ones + 1'b1;

                    if (carry_min_ones)
                        min_tens <= (min_tens == 4'd5) ? 4'd0 : min_tens + 1'b1;

                    if (carry_min_tens) begin
                        if (hour_ones == 4'd9) begin
                            hour_ones <= 4'd0;              // 09:59, 19:59 다음
                            hour_tens <= hour_tens + 1'b1;
                        end
                        else begin
                            hour_ones <= hour_ones + 1'b1;  // 일반 시간 증가
                        end
                    end
                end
            end
        end
    end

endmodule
```

## 시계 DP 점멸과 FND 모듈

**RTL 4 · DP 1Hz 점멸** — 0.5초마다 반전해 0.5초 켜짐·0.5초 꺼짐을 만듭니다. DP는 Active Low이므로 `~blink`를 내보냅니다.

파일명: `dp_blink_1hz.v`

```verilog
module dp_blink_1hz(
    input  clk,       // 100MHz 기준 Clock
    input  reset,     // 동기 Reset
    input  enable,    // Blink 동작 허용
    output dp_state   // dp 출력, Active-Low
);

    reg [25:0] count_blink;   // 0.5초 Count용
    reg        blink;         // 0.5초마다 반전되는 내부 상태

    always @(posedge clk) begin
        if (reset) begin
            count_blink <= 26'd0;
            blink       <= 1'b0;        // 초기 dp는 꺼짐
        end
        else begin
            if (enable) begin
                if (count_blink == 26'd49_999_999) begin
                    count_blink <= 26'd0;
                    blink       <= ~blink;          // 0.5초마다 반전
                end
                else begin
                    count_blink <= count_blink + 1'b1;
                end
            end
            else begin
                count_blink <= 26'd0;
                blink       <= 1'b0;                // 정지 시 dp 꺼짐
            end
        end
    end

    assign dp_state = ~blink;   // blink=1이면 dp 켜짐

endmodule
```

**RTL 5 · FND 스캔 카운터 (시계용)** — 스톱워치와 달리 **50,000클록(0.5ms)마다** 다음 자리로 넘어갑니다. 네 자리 한 바퀴가 2ms이므로 깜빡임 없이 보입니다.

파일명: `fnd_scan_counter.v`

```verilog
module fnd_scan_counter(
    input            clk,        // 기준 Clock
    input            reset,      // 동기 Reset
    output reg [1:0] digit_sel   // 현재 선택 Digit
);

    reg [15:0] scan_count;   // Scan 속도 Count

    always @(posedge clk) begin
        if (reset) begin
            scan_count <= 16'd0;
            digit_sel  <= 2'd0;
        end
        else begin
            if (scan_count == 16'd49_999) begin
                scan_count <= 16'd0;                // Scan 주기 도달
                digit_sel  <= digit_sel + 1'b1;     // 다음 Digit 선택
            end
            else begin
                scan_count <= scan_count + 1'b1;
            end
        end
    end

endmodule
```

파일명: `fnd_mux_4digit.v`

```verilog
module fnd_mux_4digit(
    input      [3:0] digit0,         // 1분 자리
    input      [3:0] digit1,         // 10분 자리
    input      [3:0] digit2,         // 1시간 자리
    input      [3:0] digit3,         // 10시간 자리
    input      [1:0] digit_sel,      // 현재 선택 Digit
    output reg [3:0] current_digit,  // Decoder로 보낼 BCD 값
    output reg [3:0] an              // FND Digit 선택, Active-Low
);

    always @(*) begin
        current_digit = 4'd0;
        an            = 4'b1111;

        case (digit_sel)
            2'd0: begin current_digit = digit0; an = 4'b1110; end   // 1분
            2'd1: begin current_digit = digit1; an = 4'b1101; end   // 10분
            2'd2: begin current_digit = digit2; an = 4'b1011; end   // 1시간
            2'd3: begin current_digit = digit3; an = 4'b0111; end   // 10시간
        endcase
    end

endmodule
```

## Basys3 24시간 시계 Top

| 포트 | 역할 |
|---|---|
| `clk` | 100MHz 클록 |
| `btnC` | Reset |
| `sw[0]` | Enable |
| `led[0]` / `led[1]` | `tick_1s` / `tick_1min` (1클록 폭이라 거의 보이지 않음) |
| `led[7:2]` | 내부 초 카운트 0~59 (2진수) |
| `seg`, `an`, `dp` | `HH.MM` 표시, 시·분 사이 DP 점멸 |

파일명: `basys3_hhmm_clock_top.v`

```verilog
module basys3_hhmm_clock_top(
    input         clk,   // Basys3 100MHz Clock
    input         btnC,  // Reset 버튼
    input  [15:0] sw,    // sw[0]: Enable
    output [15:0] led,   // 확인용 LED
    output [6:0]  seg,   // Segment 출력, Active-Low
    output [3:0]  an,    // Digit 선택, Active-Low
    output        dp     // Decimal Point, Active-Low
);

    wire       tick_1s;        // 1초 Tick
    wire       tick_1min;      // 1분 Tick
    wire [5:0] sec_count;      // 내부 초 Count
    wire [3:0] min_ones;       // 1분 자리
    wire [3:0] min_tens;       // 10분 자리
    wire [3:0] hour_ones;      // 1시간 자리
    wire [3:0] hour_tens;      // 10시간 자리
    wire [1:0] digit_sel;      // 현재 선택 Digit
    wire [3:0] current_digit;  // 현재 표시 Digit
    wire       dp_state;       // 1Hz Duty 50% dp 신호

    // LED 확인 신호
    assign led[0]    = tick_1s;
    assign led[1]    = tick_1min;
    assign led[7:2]  = sec_count;
    assign led[15:8] = 8'd0;

    // 시와 분 사이 dp만 1Hz Duty 50% 점멸
    assign dp = (digit_sel == 2'd2) ? dp_state : 1'b1;

    // 1초 Tick 생성
    clock_divider_1s u_divider(
        .clk(clk),
        .reset(btnC),
        .enable(sw[0]),
        .tick_1s(tick_1s)
    );

    // 60초 Counter
    second_counter_60 u_second(
        .clk(clk),
        .reset(btnC),
        .tick_1s(tick_1s),
        .tick_1min(tick_1min),
        .sec_count(sec_count)
    );

    // HH:MM Counter
    clock_counter_hhmm u_clock(
        .clk(clk),
        .reset(btnC),
        .tick_1min(tick_1min),
        .min_ones(min_ones),
        .min_tens(min_tens),
        .hour_ones(hour_ones),
        .hour_tens(hour_tens)
    );

    // dp 1Hz Duty 50% Blink
    dp_blink_1hz u_dp_blink(
        .clk(clk),
        .reset(btnC),
        .enable(sw[0]),
        .dp_state(dp_state)
    );

    // FND Scan Counter
    fnd_scan_counter u_scan(
        .clk(clk),
        .reset(btnC),
        .digit_sel(digit_sel)
    );

    // FND Digit 선택
    fnd_mux_4digit u_mux(
        .digit0(min_ones),
        .digit1(min_tens),
        .digit2(hour_ones),
        .digit3(hour_tens),
        .digit_sel(digit_sel),
        .current_digit(current_digit),
        .an(an)
    );

    // FND Segment Decoder
    fnd_decoder u_decoder(
        .bcd(current_digit),
        .seg(seg)
    );

endmodule
```

| 확인 항목 | 기대 동작 |
|---|---|
| `tick_1s` | 1초마다 1클록 폭 |
| `tick_1min` | 60초마다 1클록 폭 |
| HH:MM | 00:00 ~ 23:59 반복, 23:59 다음 00:00 |
| `dp_state` | 0.5초 켜짐, 0.5초 꺼짐 |
| `dp` | `digit_sel == 2'd2`일 때만 점멸 |

## 정리와 확장 과제

| 항목 | 스톱워치 | 24시간 시계 |
|---|---|---|
| 기준 Tick | 0.01초 | 1초 → 1분 |
| 증가 조건 | `sw[0] && running && tick_0p01s` | `tick_1min` |
| 자리 올림 | Carry 조건 미리 계산 | Carry + 23:59 특수 처리 |
| DP | 1초 자리 고정 점등 | 1시 자리에서 1Hz 점멸 |
| 스캔 | `scan_div[15:14]` | 50,000클록마다 자리 이동 |
| 설계 방식 | 모듈 연결형 Top | 모듈 연결형 Top |

**핵심 결론**

- Tick은 클록이 아니라 **Enable 신호**입니다.
- 모든 레지스터는 `always @(posedge clk)`에서 동작합니다.
- Top 모듈은 **배선 중심**으로 두고 기능은 재사용 가능한 하위 모듈에 둡니다. `fnd_decoder`처럼 두 프로젝트에서 그대로 재사용되는 모듈이 생깁니다.

**스스로 점검하기**

1. 스톱워치에서 125번째 Tick 후 FND에 표시되는 값은 무엇인가?

**확장 과제**

- 스톱워치를 `MM:SS`(분:초) 형식으로 바꾸고 59:59 이후 처리를 설계하시오.
- 시계에 시·분 설정 버튼을 추가하시오(버튼 디바운스·에지 검출 포함).
